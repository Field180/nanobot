# Python 客户端示例
import asyncio
import websockets
import json

async def approval_client(session_id):
    uri = f"ws://localhost:8080/ws/permission/{session_id}"
    
    async with websockets.connect(uri) as ws:
        # 订阅
        await ws.send(json.dumps({
            'type': 'subscribe',
            'session_id': session_id
        }))
        
        # 监听审批请求
        async for message in ws:
            data = json.loads(message)
            
            if data['type'] == 'permission_request':
                # 显示审批界面
                decision = await get_user_decision(data)
                
                # 发送决策
                await ws.send(json.dumps({
                    'type': 'decision',
                    'request_id': data['request_id'],
                    'decision': decision
                }))

# 运行
asyncio.run(approval_client('my-session'))
                

            

            
            

                

// React Hook 示例
import { useState, useEffect } from 'react';

function useApprovalWebSocket(sessionId) {
    const [ws, setWs] = useState(null);
    const [request, setRequest] = useState(null);
    
    useEffect(() => {
        const socket = new WebSocket(
            `ws://localhost:8080/ws/permission/${sessionId}`
        );
        
        socket.onmessage = (e) => {
            const data = JSON.parse(e.data);
            if (data.type === 'permission_request') {
                setRequest(data);
            }
        };
        
        setWs(socket);
        return () => socket.close();
    }, [sessionId]);
    
    const decide = (decision) => {
        ws?.send(JSON.stringify({
            type: 'decision',
            request_id: request.request_id,
            decision
        }));
        setRequest(null);
    };
    
    return { request, decide };
}

// 使用示例
function ApprovalDialog() {
    const { request, decide } = useApprovalWebSocket('session-123');
    
    if (!request) return null;
    
    return (
        <div className="approval-dialog">
            <h3>{request.title}</h3>
            <button onClick={() => decide('allow_once')}>允许</button>
            <button onClick={() => decide('deny')}>拒绝</button>
        </div>
    );
}
